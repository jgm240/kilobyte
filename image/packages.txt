# Packages in every Kilobyte image, on every architecture.
# Installed without "Recommends" to keep the system small.
# PC-only and Raspberry Pi-only packages are in packages-pc.txt and
# packages-pi.txt; the kernel comes from build-in-container.sh.

# --- base system ---
systemd-sysv udev dbus kmod procps iproute2 iputils-ping netbase ca-certificates
initramfs-tools busybox
sudo less nano bash-completion curl
console-setup keyboard-configuration kbd gpm

# --- network (Wi-Fi on by default), hotspot, Bluetooth ---
systemd-resolved systemd-timesyncd wpasupplicant rfkill wireless-regdb hostapd
bluez bluez-alsa-utils libasound2-plugin-bluez

# --- Kilobyte and its programs ---
dialog mc elinks sc-im calcurse htop bc bsdgames moon-buggy alpine
python3 python3-pyte weechat luit telnet openssh-client

# --- media: video, pictures, YouTube, discs, sound ---
mpv ffmpeg quickjs lsdvd eject alsa-utils alsa-ucm-conf

# --- printing (driverless: IPP Everywhere, AirPrint, IPP over USB) ---
cups cups-client cups-filters ipp-usb avahi-daemon libnss-mdns

# --- USB sticks and disks ---
fdisk e2fsprogs dosfstools exfatprogs

# --- Windows programs: Wine (per architecture, see packages-ARCH.txt) gets an
# X server of its own only while a graphical Windows program runs ---
xserver-xorg-core xserver-xorg-legacy xserver-xorg-input-libinput xserver-xorg-video-fbdev xinit

# --- network drives (Windows shares, SFTP), remote Kilobyte (SSH; the server is
# off until switched on in Settings) ---
cifs-utils sshfs openssh-server

# --- opening files from the desktop and the file manager: kinds, PDF as text ---
file poppler-utils

# --- firmware ---
firmware-linux-free firmware-realtek
